    ```swift
    func searchResultRow(_ result: FoodSearchResult) -> some View {
        Button {
            selectedFood = result
            showingFoodDetail = true
        } label: {
            FoodSearchResultRow(result: result)
        }
        .buttonStyle(.plain)
        .accessibilityIdentifier("food-result-\(result.name.lowercased().replacingOccurrences(of: " ", with: "-"))")
        .if(result.source == .userCreated) { view in
            view
                .swipeActions(edge: .trailing, allowsFullSwipe: true) {
                    Button("Delete", role: .destructive) {
                        if let food = findCustomFood(for: result) {
                            foodToDelete = food
                            showingDeleteConfirmation = true
                        }
                    }
                    .accessibilityIdentifier("delete-custom-food-button")
                }
                .swipeActions(edge: .leading, allowsFullSwipe: false) {
                    Button("Edit") {
                        if let food = findCustomFood(for: result) {
                            editingCustomFood = food
                        }
                    }
                    .tint(.blue)
                    .accessibilityIdentifier("edit-custom-food-button")
                }
        }
    }
    ```

    **4. Add helper to find Food from FoodSearchResult:**
    In FoodSearchSheet.swift, add a method to look up the actual Food entity:
    ```swift
    private func findCustomFood(for result: FoodSearchResult) -> Food? {
        // Use CustomFoodService to fetch by name (unique within custom foods)
        guard let customFoodService else { return nil }
        return try? customFoodService.getCustomFood(named: result.name)
    }
    ```

    **5. Add getCustomFood method to CustomFoodService:**
    In CustomFoodService.swift, add:
    ```swift
    func getCustomFood(named name: String) throws -> Food? {
        let descriptor = FetchDescriptor<Food>(
            predicate: #Predicate { food in
                food.source == "userCreated" && food.name == name
            }
        )
        return try context.fetch(descriptor).first
    }
    ```

    **6. Add delete confirmation alert in FoodSearchSheet.swift:**
    Add after the existing .sheet modifiers:
    ```swift
    .alert("Delete Custom Food?", isPresented: $showingDeleteConfirmation) {
        Button("Cancel", role: .cancel) {
            foodToDelete = nil
        }
        Button("Delete", role: .destructive) {
            if let food = foodToDelete {
                Task { await deleteCustomFood(food) }
            }
        }
    } message: {
        if let food = foodToDelete {
            Text("This will permanently delete '\(food.name)' from your custom foods.")
        }
    }
    ```

    **7. Add delete helper method:**
    ```swift
    private func deleteCustomFood(_ food: Food) async {
        guard let customFoodService else { return }
        do {
            try await customFoodService.deleteCustomFood(food)
            foodToDelete = nil
        } catch {
            // Log error - food may have already been deleted
        }
    }
    ```

    **8. Add edit sheet in FoodSearchSheet.swift:**
    Add after the delete alert:
    ```swift
    .sheet(item: $editingCustomFood) { food in
        if let customFoodService {
            CreateFoodSheet.edit(
                food: food,
                customFoodService: customFoodService,
                onFoodCreated: { _ in
                    editingCustomFood = nil
                }
            )
        }
    }
    ```

    **9. Add .if View extension if not already present:**
    Check if the conditional modifier exists. If not, add to a View+Extensions file:
    ```swift
    extension View {
        @ViewBuilder
        func `if`<Content: View>(_ condition: Bool, transform: (Self) -> Content) -> some View {
            if condition {
                transform(self)
            } else {
                self
            }
        }
    }
    ```

    **AVOID:** Don't add swipe actions to non-custom foods. Only .userCreated source gets edit/delete.
