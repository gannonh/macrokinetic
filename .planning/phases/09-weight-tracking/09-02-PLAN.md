    ```swift
    struct QuickWeightSheet: View {
        @Environment(\.dismiss) private var dismiss
        @Environment(\.modelContext) private var modelContext

        @State private var weightInput: String = ""
        @State private var weightUnit: String = "kg"
        @State private var bodyFatInput: String = ""
        @State private var timestamp: Date = Date()
        @State private var syncToHealthKit: Bool = true
        @State private var isSaving: Bool = false
        @State private var errorMessage: String?
        @State private var showingError: Bool = false

        private var weightService: WeightService? {
            AppServices.shared.weightService
        }

        var body: some View {
            NavigationStack {
                Form {
                    // Weight section with TextField + unit Picker
                    // Optional body fat % section
                    // Date picker (default today)
                    // HealthKit sync toggle
                }
                .navigationTitle("Log Weight")
                .toolbar {
                    ToolbarItem(placement: .cancellationAction) {
                        Button("Cancel") { dismiss() }
                    }
                    ToolbarItem(placement: .confirmationAction) {
                        Button("Save") { Task { await save() } }
                            .disabled(!canSave || isSaving)
                    }
                }
            }
            .presentationDetents([.medium])
            .onAppear { loadDefaults() }
        }
    }
    ```

    Key requirements:
    - Weight input with unit picker (kg/lbs toggle)
    - Optional body fat percentage input
    - Date picker defaulting to now
    - HealthKit sync toggle (default ON if available)
    - Validation: weight > 0, body fat 0-100 if provided
    - Convert lbs to kg before saving
    - Call weightService.syncToHealthKit() after save if toggle ON
    - Show error alert on failure
    - Accessibility identifiers: "quick-weight-sheet", "weight-input", "weight-unit-picker", "body-fat-input", "save-weight-button"

    loadDefaults() should:
    - Query latest WeightEntry for default weight value
    - Get user's preferred unit from User.weightUnit
