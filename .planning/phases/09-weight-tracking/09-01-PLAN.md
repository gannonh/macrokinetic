    Create WeightService+HealthKit.swift extension:

    ```swift
    import HealthKit

    extension WeightService {
        private static let healthStore = HKHealthStore()

        func syncToHealthKit(_ entry: WeightEntry) async throws {
            guard HKHealthStore.isHealthDataAvailable() else { return }

            // Write weight
            let weightType = HKQuantityType.quantityType(forIdentifier: .bodyMass)!
            let weightQuantity = HKQuantity(unit: .gramUnit(with: .kilo), doubleValue: entry.weightKg)
            let weightSample = HKQuantitySample(type: weightType, quantity: weightQuantity,
                                                 start: entry.timestamp, end: entry.timestamp)
            try await Self.healthStore.save(weightSample)

            // Write body fat if present
            if let bodyFat = entry.bodyFatPercentage {
                let bodyFatType = HKQuantityType.quantityType(forIdentifier: .bodyFatPercentage)!
                let bodyFatQuantity = HKQuantity(unit: .percent(), doubleValue: bodyFat / 100.0)
                let bodyFatSample = HKQuantitySample(type: bodyFatType, quantity: bodyFatQuantity,
                                                      start: entry.timestamp, end: entry.timestamp)
                try await Self.healthStore.save(bodyFatSample)
            }
        }

        var isHealthKitAvailable: Bool {
            HKHealthStore.isHealthDataAvailable()
        }
    }
    ```

    Key requirements:
    - Check isHealthDataAvailable before any operations
    - Log sync attempts and errors
    - Gracefully handle authorization denied (log warning, don't throw)
    - Body fat percentage stored as 0-100 in model, convert to 0-1 for HealthKit

    Update AppServices.swift:
    - Add `@Published private(set) var weightService: WeightService?`
    - Initialize in `initialize(with:)` method after mealLogService
    - Add to `reset()` method
  </action>
