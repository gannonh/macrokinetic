    Create unit tests for WeightService following MealLogServiceTests pattern:

    ```swift
    import Testing
    import SwiftData
    @testable import JabTracker

    @Suite("WeightService Tests")
    struct WeightServiceTests {

        private func createTestContext() -> (context: ModelContext, container: ModelContainer) {
            let schema = Schema([User.self, WeightEntry.self])
            let config = ModelConfiguration(schema: schema, isStoredInMemoryOnly: true, cloudKitDatabase: .none)
            let container = try! ModelContainer(for: schema, configurations: [config])
            return (container.mainContext, container)
        }

        @Test("Log weight creates entry")
        @MainActor
        func testLogWeight() async throws {
            let (context, container) = createTestContext()
            _ = container
            let service = WeightService(context: context)

            let entry = try service.logWeight(weightKg: 75.0, bodyFat: nil, timestamp: Date(), notes: nil)

            #expect(entry.weightKg == 75.0)
            #expect(entry.source == "manual")
        }

        @Test("Log weight validates range")
        @MainActor
        func testLogWeightValidation() async throws {
            let (context, container) = createTestContext()
            _ = container
            let service = WeightService(context: context)

            #expect(throws: WeightServiceError.self) {
                try service.logWeight(weightKg: 0, bodyFat: nil, timestamp: Date(), notes: nil)
            }
        }

        @Test("Get latest entry returns most recent")
        @MainActor
        func testGetLatestEntry() async throws {
            let (context, container) = createTestContext()
            _ = container
            let service = WeightService(context: context)

            _ = try service.logWeight(weightKg: 70.0, bodyFat: nil, timestamp: Date().addingTimeInterval(-3600), notes: nil)
            _ = try service.logWeight(weightKg: 75.0, bodyFat: nil, timestamp: Date(), notes: nil)

            let latest = try service.getLatestEntry()
            #expect(latest?.weightKg == 75.0)
        }

        @Test("Update user weight syncs latest")
        @MainActor
        func testUpdateUserWeight() async throws {
            let (context, container) = createTestContext()
            _ = container

            let user = User()
            user.weight = 70.0
            context.insert(user)

            let service = WeightService(context: context)
            _ = try service.logWeight(weightKg: 80.0, bodyFat: nil, timestamp: Date(), notes: nil)
            try service.updateUserWeight()

            #expect(user.weight == 80.0)
        }
    }
    ```

    Test cases:
    - logWeight creates entry with correct values
    - logWeight validates weight range (throws on invalid)
    - logWeight validates body fat range
    - getLatestEntry returns most recent by timestamp
    - getEntries filters by date range
    - deleteEntry removes from context
    - updateUserWeight syncs User.weight with latest entry
  </action>
