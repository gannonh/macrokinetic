/// Returns the most recent dose that was actually taken
///
/// Used to calculate next dose projections from the actual dosing pattern
/// rather than the schedule creation date.
var lastTakenDose: ScheduledDose? {
    scheduledDoses?
        .filter { $0.status == .taken }
        .max(by: { $0.scheduledTime < $1.scheduledTime })
}

Modify `getNextScheduledDose(for:)` to calculate from last actual dose when available.

**Current logic (buggy):**
```swift
func getNextScheduledDose(for schedule: DoseSchedule) -> ScheduledDose? {
    guard schedule.isActive else { return nil }
    let now = Date()
    let lookAheadDate = Calendar.current.date(byAdding: .day, value: 90, to: now)!
    let upcomingDoses = generateScheduledDoses(for: schedule, from: now, to: lookAheadDate)
    return upcomingDoses.first { $0.scheduledTime > now }
}
```

**Fixed logic:**
The issue is that `generateScheduledDoses` starts from `now` but aligns to the schedule pattern based on `schedule.createdAt`. When the user actually took doses on different dates, the projection is misaligned.

Update to calculate next dose based on last taken dose + interval:

```swift
func getNextScheduledDose(for schedule: DoseSchedule) -> ScheduledDose? {
    guard schedule.isActive else { return nil }

    let calendar = Calendar.current
    let now = Date()

    // If there's a taken dose, calculate next from that
    if let lastTaken = schedule.lastTakenDose {
        // Get the interval from config
        guard let config = try? decodeScheduleConfiguration(schedule) else {
            return nil
        }

        // Calculate next dose: last taken time + interval days
        let interval = schedule.patternType == .splitDose
            ? (config.splitIntervalMinutes ?? 0) / (24 * 60)  // Convert minutes to days
            : config.interval

        var nextDoseDate = lastTaken.scheduledTime

        // Keep adding intervals until we're past now
        while nextDoseDate <= now {
            nextDoseDate = calendar.date(byAdding: .day, value: interval, to: nextDoseDate)!
        }

        // Create the scheduled dose for this time
        let windowStart = calendar.date(byAdding: .minute, value: -config.windowMinutesBefore, to: nextDoseDate)!
        let windowEnd = calendar.date(byAdding: .minute, value: config.windowMinutesAfter, to: nextDoseDate)!

        let scheduledDose = ScheduledDose(
            scheduledTime: nextDoseDate,
            doseAmount: config.doseAmount,
            windowStart: windowStart,
            windowEnd: windowEnd
        )
        scheduledDose.schedule = schedule
        return scheduledDose
    }

    // No doses taken yet - fall back to original projection logic
    let lookAheadDate = calendar.date(byAdding: .day, value: 90, to: now)!
    let upcomingDoses = generateScheduledDoses(for: schedule, from: now, to: lookAheadDate)
    return upcomingDoses.first { $0.scheduledTime > now }
}
```

Note: The `decodeScheduleConfiguration` method is already available in this extension file.
