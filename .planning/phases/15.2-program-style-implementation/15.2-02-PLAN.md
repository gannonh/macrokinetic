    2. Update `availableSteps` computed property (around line 130) to include conditional steps:
    ```swift
    var availableSteps: [ProgramWizardStep] {
        var steps = ProgramWizardStep.allCases

        // Remove programStyle in edit mode
        if isEditMode {
            steps = steps.filter { $0 != .programStyle }
        }

        // Remove profileCompletion unless needed for Coached mode
        if !needsProfileCompletion {
            steps = steps.filter { $0 != .profileCompletion }
        }

        // Coached mode: show shiftedDaySelection when Shifted distribution selected
        // See mock: mocks/goal-program/Coached-Shift/IMG_2101-2103.PNG
        if programStyle == .coached {
            // Show shiftedDaySelection after weeklyDistribution if distribution is .shifted
            if weeklyDistribution == .shifted {
                // Insert shiftedDaySelection after weeklyDistribution
                if let idx = steps.firstIndex(of: .weeklyDistribution) {
                    steps.insert(.shiftedDaySelection, at: idx + 1)
                }
            }
        }

        // Collaborative mode: show macroCustomization after proteinLevel
        // (but NOT dietPreference, calorieFloor, proteinLevel - those are Coached-only)
        if programStyle == .collaborative {
            // Keep: programStyle, training, weeklyDistribution, macroCustomization, confirmation
            steps = steps.filter { step in
                switch step {
                case .programStyle, .training, .weeklyDistribution, .macroCustomization, .confirmation:
                    return true
                default:
                    return false
                }
            }
        }

        // Manual mode: show targetMode and conditionally singleWeekMacros or perDayMacros
        if programStyle == .manual {
            // Keep: programStyle, targetMode, [singleWeekMacros OR perDayMacros], confirmation
            steps = steps.filter { step in
                switch step {
                case .programStyle, .targetMode, .confirmation:
                    return true
                case .singleWeekMacros:
                    return useSameTargetsAllWeek
                case .perDayMacros:
                    return !useSameTargetsAllWeek
                default:
                    return false
                }
            }
        }

        return steps
    }
    ```

    ```swift
    /// Step view for selecting high calorie days in Coached/Shifted mode
    /// Mock: mocks/goal-program/Coached-Shift/IMG_2102.PNG
    struct ShiftedDaySelectionStepView: View {
        @Binding var highCalorieDays: Set<Int>

        private let days = [
            (name: "Monday", weekday: 2),
            (name: "Tuesday", weekday: 3),
            (name: "Wednesday", weekday: 4),
            (name: "Thursday", weekday: 5),
            (name: "Friday", weekday: 6),
            (name: "Saturday", weekday: 7),
            (name: "Sunday", weekday: 1)
        ]

        var body: some View {
            ScrollView {
                VStack(spacing: 24) {
                    Text("Which days would you like to have higher Calories?")
                        .font(.title2.bold())
                        .multilineTextAlignment(.leading)
                        .frame(maxWidth: .infinity, alignment: .leading)
                        .padding(.horizontal)
                        .padding(.top, 24)

                    VStack(spacing: 0) {
                        ForEach(days, id: \.weekday) { day in
                            dayRow(name: day.name, weekday: day.weekday)
                            if day.weekday != 1 {
                                Divider()
                                    .padding(.leading)
                            }
                        }
                    }
                    .padding(.horizontal)
                }
            }
            .accessibilityIdentifier("shifted-day-selection-step")
        }

        private func dayRow(name: String, weekday: Int) -> some View {
            Button {
                if highCalorieDays.contains(weekday) {
                    highCalorieDays.remove(weekday)
                } else {
                    highCalorieDays.insert(weekday)
                }
            } label: {
                HStack {
                    Text(name)
                        .foregroundColor(.primary)
                    Spacer()
                    Image(systemName: highCalorieDays.contains(weekday) ? "checkmark.square.fill" : "square")
                        .foregroundColor(highCalorieDays.contains(weekday) ? .blue : .secondary)
                        .font(.title2)
                }
                .padding(.vertical, 16)
            }
        }
    }
    ```
