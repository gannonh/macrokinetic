    2. Add configuration method:
       ```swift
       func configureRolloverProvider(mealLogService: MealLogService, modelContext: ModelContext) {
           let dataSource = MealLogNutritionDataSource(mealLogService: mealLogService, modelContext: modelContext)
           self.rolloverProvider = RolloverCalorieProvider(nutritionDataSource: dataSource)
       }
       ```

    3. Update `getAdjustedCalorieTarget()` to aggregate adjustments:
       ```swift
       func getAdjustedCalorieTarget(for user: User, on date: Date, baseTarget: Double) async -> Double {
           var adjusted = baseTarget

           // Add burned calories if enabled (existing logic)
           if user.addBurnedCaloriesEnabled {
               let activeEnergy = await getActiveEnergy(for: date)
               adjusted += activeEnergy ?? 0.0
           }

           // Add rollover if enabled (NEW)
           if let rollover = rolloverProvider {
               let bonus = await rollover.calculateAdjustment(for: user, on: date)
               adjusted += bonus
           }

           return adjusted
       }
       ```

    4. Optionally add breakdown method for future UI:
       ```swift
       struct CalorieAdjustmentBreakdown {
           let burnedCalories: Double
           let rolloverCalories: Double
           var totalAdjustment: Double { burnedCalories + rolloverCalories }
       }

       func getAdjustmentBreakdown(for user: User, on date: Date) async -> CalorieAdjustmentBreakdown
       ```

    **Do NOT** change the existing API signature - maintain backward compatibility.
