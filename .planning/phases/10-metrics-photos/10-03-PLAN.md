            #expect(entry.hipCm == nil)
            #expect(entry.chestCm == nil)

            #expect(entry.source == "manual")
            #expect(entry.notes == nil)

        @Test("Custom initialization with all parameters")
        func testCustomInitialization() { ... }

        @Test("Waist converts correctly to inches")
        func testWaistConversion() {
            let entry = MetricsEntry()
            entry.waistCm = 90.0
            #expect(entry.waistInInches != nil)
            #expect(abs(entry.waistInInches! - 35.433) < 0.01)
        }

        @Test("Hip converts correctly to inches")
        func testHipConversion() { ... }

        @Test("Chest converts correctly to inches")
        func testChestConversion() { ... }

        @Test("Neck converts correctly to inches")
        func testNeckConversion() { ... }

        @Test("Nil measurements return nil inches")
        func testNilConversion() { ... }

        // MARK: - Validation

        @Test("hasAnyMetrics true when at least one measurement present")
        func testHasAnyMetrics() {
            let entry = MetricsEntry()
            #expect(entry.hasAnyMetrics == false)
            entry.waistCm = 90.0
            #expect(entry.hasAnyMetrics == true)
        }

        @Test("isValid true for measurements in range")
        func testIsValidInRange() {
            let entry = MetricsEntry()
            entry.waistCm = 90.0
            entry.hipCm = 100.0
            #expect(entry.isValid == true)
        }

        @Test("isValid false for measurement below minimum")
        func testIsValidBelowMin() {
            let entry = MetricsEntry()
            entry.waistCm = 5.0  // Below 10 cm min
            #expect(entry.isValid == false)
        }

        @Test("isValid false for measurement above maximum")
        func testIsValidAboveMax() {
            let entry = MetricsEntry()
            entry.waistCm = 350.0  // Above 300 cm max
            #expect(entry.isValid == false)
        }

        // MARK: - Formatting

        @Test("formattedMeasurement displays metric correctly")
        func testFormattedMetric() {
            let entry = MetricsEntry()
            let result = entry.formattedMeasurement(90.0, useMetric: true)
            #expect(result == "90.0 cm")
        }

        @Test("formattedMeasurement displays imperial correctly")
        func testFormattedImperial() {
            let entry = MetricsEntry()
            let result = entry.formattedMeasurement(90.0, useMetric: false)
            #expect(result?.contains("in") == true)
        }

        // MARK: - SwiftData Persistence

        @Test("Entry persists to SwiftData")
        @MainActor
        func testPersistence() async throws {
            let (context, container) = createTestContext()
            _ = container

            let entry = MetricsEntry()
            entry.waistCm = 90.0
            entry.hipCm = 100.0
            context.insert(entry)
            try context.save()

            let descriptor = FetchDescriptor<MetricsEntry>()
            let entries = try context.fetch(descriptor)
            #expect(entries.count == 1)
            #expect(entries.first?.waistCm == 90.0)
        }
    }

    // MARK: - Test Helpers

    private func createTestContext() -> (context: ModelContext, container: ModelContainer) {
        let schema = Schema([MetricsEntry.self, ProgressPhoto.self, User.self])
        let config = ModelConfiguration(schema: schema, isStoredInMemoryOnly: true, cloudKitDatabase: .none)
        let container = try! ModelContainer(for: schema, configurations: [config])
        return (container.mainContext, container)
    }
