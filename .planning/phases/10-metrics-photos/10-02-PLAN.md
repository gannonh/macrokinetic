    **Body structure:**
    ```swift
    NavigationStack {
        Form {
            photoSection      // Photo preview or "Add Photo" button
            photoTypeSection  // Picker for front/side/back
            dateSection       // DatePicker
            notesSection      // TextField
        }
        .navigationTitle("Log Photo")
        .navigationBarTitleDisplayMode(.inline)
        .toolbar { /* Cancel, Save */ }
    }
    .presentationDetents([.large])
    .photosPicker(isPresented: $showingPhotoPicker, selection: $selectedPhotoItem, matching: .images)
    .onChange(of: selectedPhotoItem) { _, newItem in
        Task { await loadPhoto(from: newItem) }
    }
    .alert("Error", isPresented: $showingError) { ... }
    ```

    **Photo section (follow DoseEntryPhotoSection pattern):**
    ```swift
    @ViewBuilder
    private var photoSection: some View {
        Section {
            if let data = photoData, let uiImage = UIImage(data: data) {
                VStack(alignment: .leading, spacing: 8) {
                    Image(uiImage: uiImage)
                        .resizable()
                        .aspectRatio(contentMode: .fit)
                        .frame(maxHeight: 200)
                        .cornerRadius(8)
                        .accessibilityIdentifier("photo-preview")

                    Button("Change Photo") { showingPhotoPicker = true }
                        .accessibilityIdentifier("change-photo-button")

                    Button("Remove Photo", role: .destructive) {
                        photoData = nil
                        selectedPhotoItem = nil
                    }
                    .accessibilityIdentifier("remove-photo-button")
                }
            } else {
                Button("Add Photo") { showingPhotoPicker = true }
                    .accessibilityIdentifier("add-photo-button")
            }
        } header: {
            Text("Progress Photo")
        }
    }
    ```

    **Photo type section:**
    - Picker with PhotoType.allCases
    - Show icon and displayName for each option

    **loadPhoto(from:) method:**
    1. Load data via `item?.loadTransferable(type: Data.self)`
    2. Compress to JPEG: `compressImage(data)`
    3. Set photoData

    **compressImage(_:) helper:**
    ```swift
    private func compressImage(_ data: Data) -> Data? {
        guard let image = UIImage(data: data) else { return nil }
        // Target max 1MB for CloudKit
        var compression: CGFloat = 0.8
        var compressed = image.jpegData(compressionQuality: compression)
        while let data = compressed, data.count > 1_000_000, compression > 0.1 {
            compression -= 0.1
            compressed = image.jpegData(compressionQuality: compression)
        }
        return compressed
    }
    ```

    **save() method:**
    1. Guard photoData != nil
    2. Call photoService?.savePhoto(imageData:photoType:timestamp:notes:)
    3. Handle errors
    4. Dismiss on success

    **Accessibility identifiers:**
    - `quick-photo-sheet`
    - `photo-preview`, `add-photo-button`, `change-photo-button`, `remove-photo-button`
    - `photo-type-picker`
    - `date-picker`
    - `save-photo-button`, `cancel-button`
  </action>
