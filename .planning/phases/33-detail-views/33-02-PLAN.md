    private var chartSection: some View {
        VStack(alignment: .leading, spacing: 12) {
            Chart {
                ForEach(data.dailyData) { day in
                    BarMark(
                        x: .value("Date", day.date, unit: .day),
                        y: .value("Expenditure", day.value)
                    )
                    .foregroundStyle(colorForStatus(day.status))
                    .cornerRadius(2)
                }
            }
            .chartYScale(domain: 1850...2100)
            .chartXAxis {
                AxisMarks(values: .stride(by: .month)) { _ in
                    AxisValueLabel(format: .dateTime.month(.abbreviated))
                    AxisGridLine()
                }
            }
            .chartYAxis {
                AxisMarks(position: .trailing) { value in
                    AxisValueLabel()
                    AxisGridLine()
                }
            }
            .frame(height: 200)

            // Legend
            HStack(spacing: 16) {
                legendItem(color: DesignTokens.Colors.expenditure, label: "Flux Range")
                legendItem(color: .blue, label: "Updating")
                legendItem(color: .gray, label: "Holding")
            }
            .font(.caption)
        }
        .padding(16)
        .background(
            RoundedRectangle(cornerRadius: 12)
                .fill(DesignTokens.Colors.cardBackground)
        )
    }

    private func colorForStatus(_ status: ExpenditureDetailData.ExpenditureStatus) -> Color {
        switch status {
        case .fluxRange: return DesignTokens.Colors.expenditure
        case .updating: return .blue
        case .holding: return .gray
        }
    }

    ```swift
    import SwiftUI

    /// Time periods available in detail views
    enum DetailTimePeriod: String, CaseIterable {
        case oneWeek = "1W"
        case oneMonth = "1M"
        case threeMonths = "3M"
        case sixMonths = "6M"
        case oneYear = "1Y"
        case all = "All"

        var displayName: String { rawValue }
    }

    /// Reusable time period selector for detail views
    /// Includes optional detailed date toggle (D button)
    struct DetailTimePeriodSelector: View {
        @Binding var selectedPeriod: DetailTimePeriod
        @Binding var showDetailedDates: Bool

        var body: some View {
            HStack(spacing: 0) {
                // Time period buttons
                ForEach(DetailTimePeriod.allCases, id: \.self) { period in
                    Button {
                        selectedPeriod = period
                    } label: {
                        Text(period.displayName)
                            .font(.caption.weight(.medium))
                            .foregroundColor(selectedPeriod == period ? .white : .primary)
                            .padding(.horizontal, 10)
                            .padding(.vertical, 8)
                            .background(
                                RoundedRectangle(cornerRadius: 6)
                                    .fill(selectedPeriod == period ? DesignTokens.Colors.accent : Color.clear)
                            )
                    }
                }

                Divider()
                    .frame(height: 20)
                    .padding(.horizontal, 8)

                // Detailed dates toggle
                Button {
                    showDetailedDates.toggle()
                } label: {
                    Text("D")
                        .font(.caption.weight(.medium))
                        .foregroundColor(showDetailedDates ? .white : .primary)
                        .padding(.horizontal, 10)
                        .padding(.vertical, 8)
                        .background(
                            RoundedRectangle(cornerRadius: 6)
                                .fill(showDetailedDates ? DesignTokens.Colors.accent : Color.clear)
                        )
                }
            }
            .padding(4)
            .background(
                RoundedRectangle(cornerRadius: 10)
                    .fill(Color(.systemGray6))
            )
            .accessibilityIdentifier("detail-time-period-selector")
        }
    }
    ```
