    Create or update E2E test file for Food Log interactions:

    ```swift
    import XCTest

    final class FoodLogUITests: XCTestCase {
        var app: XCUIApplication!

        override func setUpWithError() throws {
            continueAfterFailure = false
            app = XCUIApplication()
            app.launchArguments = ["--ui-testing", "--seed-test-7d"]
            app.launch()
        }

        // MARK: - Tap to Edit

        /// User can tap a food entry to open edit sheet
        /// Acceptance: Tap entry → EditFoodEntrySheet opens with entry data
        func testTapFoodEntryOpensEditSheet() throws {
            // Navigate to Food Log
            app.tabBars.buttons["Food Log"].tap()

            // Wait for view to load
            let foodLogView = app.otherElements["food-log-view"]
            XCTAssertTrue(foodLogView.waitForExistence(timeout: 5))

            // Tap first food entry (seeded test data should have entries)
            // Use accessibility identifier pattern from FoodLogView
            let firstEntry = app.buttons.matching(identifier: "food-entry-row").firstMatch
            XCTAssertTrue(firstEntry.waitForExistence(timeout: 5), "Food entry should exist")
            firstEntry.tap()

            // Verify edit sheet opens
            let editSheet = app.otherElements["edit-food-entry-sheet"]
            XCTAssertTrue(editSheet.waitForExistence(timeout: 3), "Edit sheet should open on tap")
        }

        /// User can dismiss edit sheet and return to food log
        /// Acceptance: Cancel button dismisses sheet
        func testCanDismissEditSheet() throws {
            // Navigate to Food Log
            app.tabBars.buttons["Food Log"].tap()

            let foodLogView = app.otherElements["food-log-view"]
            XCTAssertTrue(foodLogView.waitForExistence(timeout: 5))

            // Tap entry to open edit sheet
            let firstEntry = app.buttons.matching(identifier: "food-entry-row").firstMatch
            XCTAssertTrue(firstEntry.waitForExistence(timeout: 5))
            firstEntry.tap()

            // Wait for edit sheet
            let editSheet = app.otherElements["edit-food-entry-sheet"]
            XCTAssertTrue(editSheet.waitForExistence(timeout: 3))

            // Tap Cancel
            app.buttons["Cancel"].tap()

            // Verify sheet dismissed
            XCTAssertFalse(editSheet.exists, "Edit sheet should be dismissed")
            XCTAssertTrue(foodLogView.exists, "Food log should still be visible")
        }
    }
    ```

    - `food-entry-row` or `food-entry-row-{uuid}` on entries
    - `edit-food-entry-sheet` on EditFoodEntrySheet
