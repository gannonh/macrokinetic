│ │     var createdAt: Date?                                                                                                                                   │ │
│ │     @Relationship(deleteRule: .cascade) var doses: [Dose]                                                                                                  │ │
│ │ }                                                                                                                                                          │ │
│ │                                                                                                                                                            │ │
│ │ @Model                                                                                                                                                     │ │
│ │ final class Dose {                                                                                                                                         │ │

│ │     var site: String?                                                                                                                                      │ │
│ │     var notes: String?                                                                                                                                     │ │
│ │     var imageData: Data?                                                                                                                                   │ │
│ │     var skipped: Bool                                                                                                                                      │ │
│ │     var user: User?                                                                                                                                        │ │
│ │     var medication: MedicationProfile?                                                                                                                     │ │
│ │ }                                                                                                                                                          │ │
│ │                                                                                                                                                            │ │

│ │ final class MedicationProfile {                                                                                                                            │ │
│ │     var id: UUID                                                                                                                                           │ │
│ │     var genericName: String?                                                                                                                               │ │
│ │     var brandName: String?                                                                                                                                 │ │
│ │     var currentDose: Double                                                                                                                                │ │
│ │     var startDate: Date?                                                                                                                                   │ │
│ │     var refillDate: Date?                                                                                                                                  │ │

│ │ Phase 3: Replace PersistenceController                                                                                                                     │ │
│ │                                                                                                                                                            │ │
│ │ 1. Create DataController.swift                                                                                                                             │ │
│ │   - Replace NSPersistentContainer with ModelContainer                                                                                                      │ │
│ │   - Use ModelContext instead of NSManagedObjectContext                                                                                                     │ │
│ │   - Implement CloudKit sync configuration                                                                                                                  │ │
│ │ 2. Structure:                                                                                                                                              │ │
│ │ @MainActor                                                                                                                                                 │ │
│ │ class DataController: ObservableObject {                                                                                                                   │ │
│ │     static let shared = DataController()                                                                                                                   │ │
│ │     let container: ModelContainer                                                                                                                          │ │
│ │                                                                                                                                                            │ │
│ │     init(inMemory: Bool = false) {                                                                                                                         │ │
│ │         let schema = Schema([User.self, Dose.self, MedicationProfile.self])                                                                                │ │
│ │         let config = ModelConfiguration(                                                                                                                   │ │
│ │             schema: schema,                                                                                                                                │ │
│ │             isStoredInMemoryOnly: inMemory,                                                                                                                │ │
│ │             cloudKitDatabase: inMemory ? .none : .automatic                                                                                                │ │
│ │         )                                                                                                                                                  │ │
│ │         container = try! ModelContainer(for: schema, configurations: [config])                                                                             │ │
│ │     }                                                                                                                                                      │ │
│ │ }                                                                                                                                                          │ │
