// JabTrackerUITests/ScheduleManagementUITests.swift
final class ScheduleManagementUITests: XCTestCase {
    var app: XCUIApplication!

    override func setUp() {
        super.setUp()
        continueAfterFailure = false
        app = TestUtilities.launchAppWithTestMode()
    }

    func testCreateWeeklySchedule() throws {
        // GIVEN: User is on Settings → Schedule Management
        TestUtilities.navigateToSettings(in: app)
        app.buttons["Schedule Management"].tap()

        // WHEN: User creates weekly schedule
        app.buttons["add-schedule-button"].tap()

        // Select weekly schedule type
        app.pickers["schedule-type-picker"].tap()
        app.pickerWheels.element.adjust(toPickerWheelValue: "Weekly")

        // Select day of week
        app.pickers["day-of-week-picker"].tap()
        app.pickerWheels.element.adjust(toPickerWheelValue: "Sunday")

        // Set preferred dose time
        let timeField = app.textFields["preferred-time-field"]
        TestUtilities.clearAndEnterText(in: timeField, newText: "09:00 AM")

        app.buttons["save-schedule-button"].tap()

        // THEN: Schedule appears in list with correct details
        let scheduleRow = app.staticTexts["Sunday at 09:00 AM"]
        XCTAssertTrue(scheduleRow.waitForExistence(timeout: 5))
    }

    func testUpdateSchedule() throws
    func testDeleteSchedule() throws
    func testScheduleFormValidation() throws
    func testScheduleConflictDetection() throws
}

### Test Data Seeding for E2E Tests

```swift
// JabTrackerUITests/Utils/TestUtilities+ScheduleSeeding.swift
extension TestUtilities {
    enum SchedulePreset {
        case weeklySchedule // Sunday 9am, 30 days history
        case dailySchedule // 10am daily, 14 days history
        case splitDoseSchedule // 9am + 9pm, 14 days history
        case customSchedule // Mon/Wed/Fri 8am, 30 days history
    }

    static func launchAppWithSchedulePreset(_ preset: SchedulePreset) -> XCUIApplication {
        let app = XCUIApplication()
        app.launchArguments = [
            "--ui-testing",
            "--reset-app-data",
            "--seed-schedule-\(preset.rawValue)"
        ]
        app.launch()
        return app
    }
}
```
