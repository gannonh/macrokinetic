```swift
struct ScheduleSetupView: View {
    @Bindable var viewModel: OnboardingViewModel
    @State private var selectedPattern: SchedulePattern = .weekly
    @State private var reminderMinutes: Int = 30
    @State private var enableMultipleReminders: Bool = false
    @State private var showConcentrationPreview: Bool = true

    var body: some View {
        VStack(spacing: 24) {
            // Header
            OnboardingHeaderView(
                title: "Set Up Your Dose Schedule",
                subtitle: "Choose when you'll take your medication"
            )

            // Pattern Selection
            SchedulePatternPicker(
                selectedPattern: $selectedPattern,
                onPatternChange: updateConcentrationPreview
            )

            // Concentration Curve Preview
            if showConcentrationPreview {
                ConcentrationCurvePreview(
                    pattern: selectedPattern,
                    medication: viewModel.selectedMedication,
                    pkEngine: viewModel.pkEngine
                )
                .frame(height: 200)
            }

            // Reminder Configuration
            ReminderPreferencesView(
                reminderMinutes: $reminderMinutes,
                enableMultiple: $enableMultipleReminders
            )

            Spacer()

            // Continue Button
            PrimaryButton("Continue") {
                viewModel.saveScheduleConfiguration(
                    pattern: selectedPattern,
                    reminderMinutes: reminderMinutes,
                    enableMultiple: enableMultipleReminders
                )
            }
            .disabled(!isConfigurationValid)
        }
        .padding()
    }

    private var isConfigurationValid: Bool {
        // Validate schedule configuration
        selectedPattern != .custom || viewModel.customScheduleValid
    }
}
```

#### 3. ConcentrationCurvePreview
**Location**: `JabTracker/Views/Onboarding/Components/ConcentrationCurvePreview.swift`

**Responsibilities**:
- Use PharmacokineticsEngine to calculate concentration curves
- Display concentration over time for selected pattern
- Show peak/trough levels with annotations
- Compare different patterns (optional)

**Implementation**:
```swift
struct ConcentrationCurvePreview: View {
    let pattern: SchedulePattern
    let medication: Medication?
    let pkEngine: PharmacokineticsEngine

    @State private var concentrationData: [ConcentrationPoint] = []

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            Text("How Your Medication Works")
                .font(.headline)

            Chart {
                ForEach(concentrationData) { point in
                    LineMark(
                        x: .value("Time", point.date),
                        y: .value("Concentration", point.concentration)
                    )
                    .foregroundStyle(.blue)
                }
            }
            .chartYScale(domain: 0...maxConcentration * 1.1)
            .frame(height: 150)

            HStack {
                ConcentrationLabel(
                    title: "Peak",
                    value: peakConcentration,
                    color: .green
                )
                Spacer()
                ConcentrationLabel(
                    title: "Trough",
                    value: troughConcentration,
                    color: .orange
                )
            }
        }
        .padding()
        .background(
            RoundedRectangle(cornerRadius: 12)
                .fill(Color.gray.opacity(0.05))
        )
        .task {
            await updateConcentrationData()
        }
        .onChange(of: pattern) { _, _ in
            Task { await updateConcentrationData() }
        }
    }

    @MainActor
    private func updateConcentrationData() async {
        guard let medication = medication else { return }

        // Generate sample schedule based on pattern
        let sampleSchedule = createSampleSchedule(pattern: pattern, medication: medication)

        // Calculate concentration curve
        concentrationData = pkEngine.calculateConcentrationCurve(
            schedule: sampleSchedule,
            days: 30
        )
    }
}
```

```swift
extension OnboardingViewModel {
    /// Save schedule configuration and create initial schedule
    func saveScheduleConfiguration(
        pattern: SchedulePattern,
        reminderMinutes: Int,
        enableMultiple: Bool
    ) {
        self.schedulePattern = pattern
        self.reminderMinutes = reminderMinutes
        self.enableMultipleReminders = enableMultiple

        // Validate configuration
        guard validateScheduleConfiguration() else {
            // Show error message
            return
        }

        // Proceed to next onboarding step
        moveToNextStep()
    }

    /// Create initial dose schedule after onboarding completion
    @MainActor
    func createInitialSchedule() async throws {
        guard let medicationProfile = selectedMedicationProfile,
              let pattern = schedulePattern else {
            throw OnboardingError.missingScheduleConfiguration
        }

        // Create schedule configuration
        let config = ScheduleConfig(
            pattern: pattern,
            frequency: pattern.defaultFrequency,
            reminderMinutesBefore: reminderMinutes,
            enableMultipleReminders: enableMultipleReminders
        )

        // Create schedule via ScheduleService
        let schedule = try scheduleService.createSchedule(
            for: medicationProfile,
            config: config
        )

        // Schedule notifications via NotificationService
        try await notificationService.scheduleNotifications(for: schedule)
    }

    /// Validate schedule configuration is complete and valid
    private func validateScheduleConfiguration() -> Bool {
        guard schedulePattern != nil else { return false }

        // Custom pattern validation
        if schedulePattern == .custom {
            return customScheduleValid
        }

        return true
    }
}
```

## Implementation Notes

### Technical Considerations
1. **PharmacokineticsEngine Integration**: Reuse existing engine for concentration curve calculations. Generate sample dose schedule based on pattern selection to feed into engine.

2. **Concentration Curve Performance**: Pre-calculate concentration curves for standard patterns during view initialization. Only recalculate on pattern change, not on every render.

3. **Notification Permission Timing**: Request notification permission during this step (after schedule configuration). Handle permission denial gracefully with in-app reminder fallback.

4. **Schedule Validation**: Weekly pattern always valid. Split-dose pattern requires frequency validation. Custom pattern requires user to configure at least 1 scheduled dose.

5. **Data Persistence**: Save schedule configuration to OnboardingViewModel state. Only create DoseSchedule entity after onboarding completion to avoid partial data.

6. **Error Handling**: If schedule creation fails during onboarding completion, allow user to skip schedule setup and configure later via medication profile.

### Reusable Components
- **OnboardingHeaderView**: Existing component for step titles
- **PrimaryButton**: Existing design system button
- **PharmacokineticsEngine**: Existing service for concentration calculations
- **ScheduleService**: Task 002 service for schedule CRUD
- **NotificationService**: Task 003 service for notification scheduling

### Edge Cases
1. **No Medication Selected**: Schedule setup requires medication selection. Prevent access if user navigates back from dose entry step.
2. **Custom Pattern Complexity**: Phase 1 implementation: Custom pattern shows "Coming Soon" message, defaults to weekly pattern.
3. **Notification Permission Denied**: Show warning that in-app schedule view will be primary reminder method.
4. **Onboarding Exit Mid-Setup**: Persist schedule configuration in OnboardingViewModel. Restore state if user returns.

### Accessibility Requirements
- All schedule pattern cards keyboard navigable
- Concentration curve preview has accessibility label describing peak/trough levels
- Reminder time picker announces selected value
- Continue button disabled state announced with reason ("Select a schedule pattern to continue")

## Learnings & Knowledge Capture

### Technical Patterns → system-patterns.md

#### E2E Testing Patterns
- **Debug-First Element Discovery**: Always use `TestUtilities.debugElements()` before writing element selectors to understand actual accessibility hierarchy
- **Multiple Element Handling**: Use `.element(boundBy: 0)` pattern for accessing specific elements when multiple share the same identifier
- **Back Button Access Pattern**: Use explicit accessibility identifier (`app.buttons["onboarding-back-button"]`) rather than navigation bar queries
- **Integration Test Patterns**: Handle optional screens (notifications) with `waitForExistence(timeout:)` and conditional logic
- **Tab Bar Verification**: Tab bar existence sufficient for main app verification in integration tests
- **Strategic Sleep Placement**: Use `sleep(3)` for navigation and rendering, `usleep(50_000)` for UI update timing

#### Performance Testing Patterns
- **E2E Performance Expectations**: E2E tests require < 1 second timeouts (not < 200ms unit test expectations)
- **Measurement Patterns**: Use `Date()` timestamps for performance measurement, `usleep()` for small delays
- **Chart Preview Performance**: Pattern change validation achieves < 1 second requirement

#### Accessibility Testing Patterns
- **VoiceOver Validation Without Simulation**: Validate accessibility properties (labels, values, hints) without simulating actual VoiceOver
