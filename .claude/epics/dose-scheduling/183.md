### Edge Case Handling

#### Timezone & DST Edge Cases
```swift
// Edge Case 1: User travels across timezones
// Solution: Store scheduled dose times in user's current timezone
// Behavior: Schedule adjusts to local time, notifications reschedule

// Edge Case 2: Daylight Saving Time transitions
// Solution: Use Calendar-based date calculations (not TimeInterval math)
// Validation: Test with DST transition dates (spring forward, fall back)

@Test("Schedule handles DST spring forward correctly")
func testDST_SpringForward() {
    // March 10, 2024: 2:00 AM → 3:00 AM (loses 1 hour)
    // Schedule at 2:30 AM should move to 3:30 AM
    let springForwardDate = Calendar.current.date(from: DateComponents(
        year: 2024, month: 3, day: 10, hour: 1, minute: 0
    ))!

    let schedule = ScheduleModel(
        type: .daily,
        preferredDoseTime: DateComponents(hour: 2, minute: 30)
    )

    let nextDose = scheduleService.calculateNextScheduledDose(
        schedule: schedule,
        after: springForwardDate
    )

    // Verify next dose is at 3:30 AM (adjusted for DST)
    #expect(nextDose.hour == 3)
    #expect(nextDose.minute == 30)
}
```

### Production Readiness Checklist

#### Error Handling & Recovery
```swift
// Comprehensive error handling for all schedule operations
enum ScheduleError: LocalizedError {
    case invalidTimeConfiguration
    case notificationSchedulingFailed(Error)
    case scheduleConflict
    case timezoneConversionFailed

    var errorDescription: String? {
        switch self {
        case .invalidTimeConfiguration:
            return "Invalid schedule time. Please check your dose times and try again."
        case .notificationSchedulingFailed(let error):
            return "Could not schedule notifications: \(error.localizedDescription)"
        case .scheduleConflict:
            return "This medication already has a schedule. Edit or delete the existing schedule first."
        case .timezoneConversionFailed:
            return "Could not determine your timezone. Please check your device settings."
        }
    }

    var recoverySuggestion: String? {
        switch self {
        case .invalidTimeConfiguration:
            return "Ensure dose times are valid and split-dose times are at least 4 hours apart."
        case .notificationSchedulingFailed:
            return "Check notification permissions in Settings → Notifications → JabTracker."
        case .scheduleConflict:
            return "Go to Schedule Management to edit or delete your existing schedule."
        case .timezoneConversionFailed:
            return "Go to Settings → General → Date & Time and ensure timezone is set correctly."
        }
    }
}
```
