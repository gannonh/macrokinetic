## Technical Approach

### 1. AnalyticsService Integration Layer

Create bridge between ScheduleService and Analytics UI:

```swift
// JabTracker/Services/AnalyticsService+Scheduling.swift
extension AnalyticsService {
    /// Get schedule adherence metrics for display in Analytics UI
    /// Wraps ScheduleService.calculateAdherence() with UI-friendly transformations
    func getScheduleAdherenceForDisplay(
        profile: MedicationProfile,
        dateRange: ClosedRange<Date>,
        context: ModelContext
    ) async -> DisplayableScheduleMetrics?

    /// Get upcoming scheduled doses for timeline display
    /// Wraps ScheduleService.generateScheduledDoses() with chart-compatible format
    func getUpcomingScheduledDoses(
        profile: MedicationProfile,
        context: ModelContext,
        days: Int = 30
    ) async -> [ScheduledDoseMarker]

    /// Project concentration curve based on scheduled doses
    /// Combines ScheduleService projection with PharmacokineticsEngine
    func projectConcentrationFromSchedule(
        profile: MedicationProfile,
        context: ModelContext,
        projectionDays: Int = 90
    ) async -> [ProjectedConcentrationPoint]
}

struct DisplayableScheduleMetrics {
    let adherencePercentage: Double
    let adherenceRating: String // "Excellent", "Good", "Needs Improvement"
    let currentStreak: Int
    let longestStreak: Int
    let totalScheduled: Int
    let totalTaken: Int
    let totalMissed: Int
    let onTimePercentage: Double
    let insights: [DisplayableInsight]
}

struct DisplayableInsight {
    let icon: String
    let title: String
    let message: String
    let severity: InsightSeverity
    let recommendation: String?
}

struct ScheduledDoseMarker {
    let date: Date
    let amount: Double
    let isUpcoming: Bool // true for future, false for past scheduled
}

struct ProjectedConcentrationPoint {
    let date: Date
    let concentration: Double
    let isProjected: Bool // true for future projection
}
```
